cmake_minimum_required(VERSION 3.21)
set(CMAKE_CONFIGURATION_TYPES Release CACHE STRING "Supported configurations" FORCE)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
project(XenonBrowser VERSION 0.1.1 LANGUAGES C CXX)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/VERSION")
file(STRINGS "${CMAKE_SOURCE_DIR}/VERSION" XENON_VERSION LIMIT_COUNT 1)
if(NOT XENON_VERSION MATCHES "^([0-9]+)[.]([0-9]+)[.]([0-9]+)(-alpha[.]([0-9]+))?$")
  message(FATAL_ERROR "VERSION must be a release or numbered alpha version")
endif()
set(XENON_VERSION_BUILD 0)
if(CMAKE_MATCH_5)
  set(XENON_VERSION_BUILD "${CMAKE_MATCH_5}")
endif()
set(XENON_VERSION_NUMERIC "${CMAKE_MATCH_1}.${CMAKE_MATCH_2}.${CMAKE_MATCH_3}.${XENON_VERSION_BUILD}")
string(REPLACE "." "," XENON_VERSION_RESOURCE "${XENON_VERSION_NUMERIC}")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set_property(GLOBAL PROPERTY USE_FOLDERS ON)
include(CTest)
option(XENON_BUILD_BROWSER "Build the CEF browser application" ON)
option(XENON_BUILD_AUTH_FIXTURE "Build test-only browser trusting the pinned localhost fixture certificate" OFF)
set(CEF_ROOT "${CMAKE_SOURCE_DIR}/third_party/cef" CACHE PATH "Pinned CEF distribution")

add_library(nlohmann_json INTERFACE)
add_library(nlohmann_json::nlohmann_json ALIAS nlohmann_json)
target_include_directories(nlohmann_json SYSTEM INTERFACE "${CMAKE_SOURCE_DIR}/third_party/json")
add_library(sqlite3 STATIC third_party/sqlite/sqlite3.c)
add_library(SQLite::SQLite3 ALIAS sqlite3)
target_include_directories(sqlite3 PUBLIC third_party/sqlite)
target_compile_definitions(sqlite3 PRIVATE SQLITE_THREADSAFE=1 SQLITE_DQS=0 SQLITE_OMIT_LOAD_EXTENSION=1 SQLITE_DEFAULT_MEMSTATUS=0 SQLITE_SECURE_DELETE=1)

file(GLOB XENON_CORE_SOURCES CONFIGURE_DEPENDS native/src/broker.cpp native/src/browser_data.cpp native/src/pipe_server.cpp native/src/vault.cpp native/src/file_policy.cpp native/src/local_security.cpp native/src/visible_evidence.cpp native/src/workspace_storage.cpp native/src/updater.cpp)
add_library(xenon_core STATIC ${XENON_CORE_SOURCES})
target_include_directories(xenon_core PUBLIC native/include)
target_compile_definitions(xenon_core PUBLIC XENON_VERSION="${XENON_VERSION}")
target_link_libraries(xenon_core PUBLIC nlohmann_json::nlohmann_json SQLite::SQLite3)
if(WIN32)
  target_compile_definitions(xenon_core PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  target_link_libraries(xenon_core PUBLIC bcrypt advapi32 crypt32 winhttp ws2_32 shell32 ole32)
  target_compile_options(xenon_core PRIVATE /W4 /EHsc /utf-8)
endif()

if(BUILD_TESTING)
  if(WIN32)
    add_executable(updater_launch_fixture WIN32 tests/native/updater_launch_fixture.cpp)
    target_compile_definitions(updater_launch_fixture PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_compile_options(updater_launch_fixture PRIVATE /EHsc /utf-8)
  endif()
  if(EXISTS "${CMAKE_SOURCE_DIR}/tests/native/vault_fixture_seed.cpp")
    add_executable(vault_fixture_seed tests/native/vault_fixture_seed.cpp)
    target_link_libraries(vault_fixture_seed PRIVATE xenon_core)
    target_compile_options(vault_fixture_seed PRIVATE /EHsc /utf-8)
  endif()
  file(GLOB XENON_TEST_SOURCES CONFIGURE_DEPENDS tests/native/*_tests.cpp)
  foreach(test_source IN LISTS XENON_TEST_SOURCES)
    get_filename_component(test_name "${test_source}" NAME_WE)
    add_executable(${test_name} "${test_source}")
    target_link_libraries(${test_name} PRIVATE xenon_core)
    if(WIN32 AND test_name STREQUAL "updater_tests")
      add_dependencies(updater_tests updater_launch_fixture)
    endif()
    if(WIN32 AND test_name STREQUAL "native_ui_tests")
      target_sources(${test_name} PRIVATE native/src/introduction.cpp)
      target_link_libraries(${test_name} PRIVATE comctl32 uxtheme dwmapi gdiplus)
    endif()
    if(MSVC)
      target_compile_options(${test_name} PRIVATE /UNDEBUG /EHsc /utf-8)
    endif()
    add_test(NAME ${test_name} COMMAND ${test_name})
  endforeach()
endif()

if(XENON_BUILD_BROWSER)
  if(NOT WIN32)
    message(FATAL_ERROR "The browser package currently targets Windows x64.")
  endif()
  enable_language(RC)
  configure_file(native/resources/xenon.rc.in "${CMAKE_BINARY_DIR}/xenon.rc" @ONLY)
  add_executable(xenon_brand_bootstrap tools/brand_bootstrap.cpp)
  target_compile_definitions(xenon_brand_bootstrap PRIVATE UNICODE _UNICODE)
  target_compile_options(xenon_brand_bootstrap PRIVATE /EHsc /utf-8)
  set(USE_SANDBOX ON CACHE BOOL "Sandbox is mandatory" FORCE)
  set(USE_ATL OFF CACHE BOOL "Native Win32 UI does not require ATL" FORCE)
  list(APPEND CMAKE_MODULE_PATH "${CEF_ROOT}/cmake")
  find_package(CEF REQUIRED)
  # Catch all app exceptions at callback boundaries; never cross the CEF ABI.
  list(REMOVE_ITEM CEF_COMPILER_DEFINES "_HAS_EXCEPTIONS=0")
  list(APPEND CEF_COMPILER_DEFINES "_HAS_EXCEPTIONS=1")
  add_subdirectory("${CEF_LIBCEF_DLL_WRAPPER_PATH}" libcef_dll_wrapper)
  ADD_LOGICAL_TARGET(libcef_lib "${CEF_LIB_RELEASE}" "${CEF_LIB_RELEASE}")
  file(GLOB XENON_BROWSER_SOURCES CONFIGURE_DEPENDS native/src/cef_*.cpp native/src/main_win.cpp native/src/native_ui.cpp native/src/introduction.cpp native/resources/*.rc)
  add_library(Xenon SHARED ${XENON_BROWSER_SOURCES} "${CMAKE_BINARY_DIR}/xenon.rc")
  add_dependencies(Xenon xenon_brand_bootstrap)
  SET_LIBRARY_TARGET_PROPERTIES(Xenon)
  target_compile_options(Xenon PRIVATE /EHsc /WX- /utf-8)
  target_include_directories(Xenon PRIVATE "${CEF_ROOT}" native/include)
  target_link_libraries(Xenon PRIVATE xenon_core libcef_lib libcef_dll_wrapper ${CEF_STANDARD_LIBS} comdlg32 comctl32 dwmapi uxtheme gdiplus ole32 shell32 wtsapi32)
  set_target_properties(Xenon PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/app" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/app")
  add_custom_command(TARGET Xenon POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CEF_ROOT}/Release" "$<TARGET_FILE_DIR:Xenon>"
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CEF_ROOT}/Resources" "$<TARGET_FILE_DIR:Xenon>"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CEF_ROOT}/Release/bootstrap.exe" "$<TARGET_FILE_DIR:Xenon>/Xenon.exe"
    COMMAND $<TARGET_FILE:xenon_brand_bootstrap> "${CEF_ROOT}/Release/bootstrap.exe" "$<TARGET_FILE_DIR:Xenon>/Xenon.exe" "$<TARGET_FILE:Xenon>"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CEF_ROOT}/LICENSE.txt" "$<TARGET_FILE_DIR:Xenon>/CEF-LICENSE.txt"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CEF_ROOT}/CREDITS.html" "$<TARGET_FILE_DIR:Xenon>/Chromium-CREDITS.html"
    VERBATIM)
  if(XENON_BUILD_AUTH_FIXTURE)
    add_library(XenonAuthTest SHARED ${XENON_BROWSER_SOURCES} "${CMAKE_BINARY_DIR}/xenon.rc")
    add_dependencies(XenonAuthTest xenon_brand_bootstrap)
    SET_LIBRARY_TARGET_PROPERTIES(XenonAuthTest)
    target_compile_options(XenonAuthTest PRIVATE /EHsc /WX- /utf-8)
    target_compile_definitions(XenonAuthTest PRIVATE XENON_TEST_FIXTURE_CERT_SHA256="ebd78c975821dd1c14acb73a864cc5d7641fb5af626a4a8ad6cca54692495e06")
    target_include_directories(XenonAuthTest PRIVATE "${CEF_ROOT}" native/include)
    target_link_libraries(XenonAuthTest PRIVATE xenon_core libcef_lib libcef_dll_wrapper ${CEF_STANDARD_LIBS} comdlg32 comctl32 dwmapi uxtheme gdiplus ole32 shell32 wtsapi32)
    set_target_properties(XenonAuthTest PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/auth-fixture" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/auth-fixture")
    add_custom_command(TARGET XenonAuthTest POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E copy_directory "${CEF_ROOT}/Release" "$<TARGET_FILE_DIR:XenonAuthTest>"
      COMMAND ${CMAKE_COMMAND} -E copy_directory "${CEF_ROOT}/Resources" "$<TARGET_FILE_DIR:XenonAuthTest>"
      COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CEF_ROOT}/Release/bootstrap.exe" "$<TARGET_FILE_DIR:XenonAuthTest>/XenonAuthTest.exe"
      COMMAND $<TARGET_FILE:xenon_brand_bootstrap> "${CEF_ROOT}/Release/bootstrap.exe" "$<TARGET_FILE_DIR:XenonAuthTest>/XenonAuthTest.exe" "$<TARGET_FILE:XenonAuthTest>"
      VERBATIM)
  endif()
endif()
